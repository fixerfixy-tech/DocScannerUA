package com.fixerfixy.docscanner

import android.content.Context
import android.graphics.Bitmap
import android.util.Log
import com.googlecode.tesseract.android.TessBaseAPI
import java.io.File
import java.io.FileOutputStream

object OcrManager {

    @Volatile
    private var isInitialized = false
    private var initError: String? = null
    private val tessBaseAPI = TessBaseAPI()

    @Synchronized
    fun init(context: Context): Boolean {
        if (isInitialized) return true

        try {
            val tessDir = File(context.filesDir, "tessdata")
            if (!tessDir.exists()) {
                tessDir.mkdirs()
            }

            val ukrFile = File(tessDir, "ukr.traineddata")
            copyAssetIfNeeded(context, "tessdata/ukr.traineddata", ukrFile)

            val dataPath = context.filesDir.absolutePath + File.separator

            val success = tessBaseAPI.init(dataPath, "ukr")
            if (success) {
                tessBaseAPI.pageSegMode = TessBaseAPI.PageSegMode.PSM_AUTO
                isInitialized = true
                initError = null
                return true
            } else {
                initError = "Не вдалося ініціалізувати Tesseract для шляху: $dataPath"
                return false
            }
        } catch (e: Exception) {
            initError = "Помилка init: ${e.localizedMessage}"
            Log.e("OcrManager", initError, e)
            return false
        }
    }

    fun extractText(context: Context, bitmap: Bitmap): String {
        if (!isInitialized) {
            val ready = init(context)
            if (!ready) {
                return "[Помилка ініціалізації OCR: ${initError ?: "невідома"}]"
            }
        }

        return synchronized(this) {
            try {
                val argbBmp = if (bitmap.config != Bitmap.Config.ARGB_8888) {
                    bitmap.copy(Bitmap.Config.ARGB_8888, false)
                } else {
                    bitmap
                }
                val cleanBmp = removeBlueStampsForOcr(argbBmp)
                tessBaseAPI.setImage(cleanBmp)
                val text = tessBaseAPI.utF8Text ?: ""
                tessBaseAPI.clear()
                if (text.isBlank()) "[Текст не знайдено на сторінці]" else text
            } catch (e: Exception) {
                "[Помилка читання OCR: ${e.localizedMessage}]"
            }
        }
    }

    private fun removeBlueStampsForOcr(src: Bitmap): Bitmap {
        val width = src.width
        val height = src.height
        val pixels = IntArray(width * height)
        src.getPixels(pixels, 0, width, 0, 0, width, height)

        for (i in pixels.indices) {
            val c = pixels[i]
            val r = (c shr 16) and 0xFF
            val g = (c shr 8) and 0xFF
            val b = c and 0xFF

            // Вибілюємо лише насичені сині печатки (b - r > 45), зберігаючи чорний і темно-сірий текст
            val gray = if (b - r > 45 && b > 100 && r < 140) {
                255
            } else {
                (r * 77 + g * 150 + b * 29) shr 8
            }
            pixels[i] = (0xFF shl 24) or (gray shl 16) or (gray shl 8) or gray
        }

        val out = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        out.setPixels(pixels, 0, width, 0, 0, width, height)
        return out
    }

    private fun copyAssetIfNeeded(context: Context, assetPath: String, destFile: File) {
        if (destFile.exists() && destFile.length() > 100_000) return
        try {
            context.assets.open(assetPath).use { input ->
                FileOutputStream(destFile).use { output ->
                    input.copyTo(output)
                }
            }
        } catch (ignored: Exception) {}
    }
}

data class ParsedDocument(

    var number: String,
    var date: String,
    val rawOcrText: String,
    val contractNumber: String = "",
    val contractDate: String = ""
) {
    fun buildStandardFileName(extension: String = "pdf"): String {
        val safeNum = number.ifBlank { "б-н" }.replace('/', '-')
        val safeDate = date.ifBlank { "не_вказано" }
        val ext = extension.lowercase().removePrefix(".")

        return if ((type == "Специфікація" || type == "Додаткова угода") && contractNumber.isNotBlank()) {
            val safeContNum = contractNumber.replace('/', '-')
            val safeContDate = contractDate.ifBlank { safeDate }
            "${type}_№${safeNum}_від_${safeDate}_до_Договору_№${safeContNum}_від_${safeContDate}_р.$ext"
        } else {
            val safeNumClean = safeNum.replace("№", "")
            "${type}_No_${safeNumClean}_від_${safeDate}_р.$ext"
        }
    }
}
### AppPreferences
package com.fixerfixy.docscanner

import android.content.Context
import android.content.SharedPreferences
import android.net.Uri
import android.os.Environment
import java.io.File

object AppPreferences {
    private const val PREFS_NAME = "doc_scanner_settings"
    private const val KEY_FORMAT = "pref_format"
    private const val KEY_COLOR = "pref_color"
    private const val KEY_DPI = "pref_dpi"
    private const val KEY_CUSTOM_URI = "pref_custom_uri"
    private const val KEY_FOLDER_NAME = "pref_folder_name"

    fun getFormat(context: Context): String =
        getPrefs(context).getString(KEY_FORMAT, "pdf") ?: "pdf"

    fun setFormat(context: Context, format: String) {
        getPrefs(context).edit().putString(KEY_FORMAT, format).apply()
    }

    fun getColorMode(context: Context): String =
        getPrefs(context).getString(KEY_COLOR, "color") ?: "color"

    fun setColorMode(context: Context, mode: String) {
        getPrefs(context).edit().putString(KEY_COLOR, mode).apply()
    }

    fun getDpi(context: Context): Int =
        getPrefs(context).getInt(KEY_DPI, 200)

    fun setDpi(context: Context, dpi: Int) {
        getPrefs(context).edit().putInt(KEY_DPI, dpi).apply()
    }

    fun getCustomTreeUri(context: Context): Uri? {
        val uriStr = getPrefs(context).getString(KEY_CUSTOM_URI, null)
        return uriStr?.let { Uri.parse(it) }
    }

    fun setCustomTreeUri(context: Context, uri: Uri?, displayName: String?) {
        getPrefs(context).edit()
            .putString(KEY_CUSTOM_URI, uri?.toString())
            .putString(KEY_FOLDER_NAME, displayName)
            .apply()
    }

    fun getFolderDisplayName(context: Context): String {
        return getPrefs(context).getString(KEY_FOLDER_NAME, null)
            ?: "Внутрішня пам'ять / Documents / ScanDoc"
    }

    fun getDefaultFolder(): File {
        return File(Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOCUMENTS), "ScanDoc")
    }

    private fun getPrefs(context: Context): SharedPreferences =
        context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
}
